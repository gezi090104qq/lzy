	local function translate(obj)
	    if not (obj:IsA("TextLabel") or obj:IsA("TextButton") or obj:IsA("TextBox")) then
	        return
	    end
	    local raw = obj.Text
	    if raw == "" then return end
	    local t = raw:match("^%s*(.-)%s*$") or raw
	    -- 第1步：精确匹配
	    local replacement = ManualDictionary[t]
	    if replacement then
	        obj.Text = replacement
	        return
	    end
	    -- 第2步：短语替换（下拉框组合文本）
	    local result = t
	    local changed = false
	    for en, zh in pairs(Phrases) do
	        local r = result:gsub(en, zh)
	        if r ~= result then
	            changed = true
	            result = r
	        end
	    end
	    if changed then
	        obj.Text = result
	    end
	end
	local function setupSilentTranslator(parent)
	    for _, descendant in pairs(parent:GetDescendants()) do
	        translate(descendant)
	    end
	    parent.DescendantAdded:Connect(function(descendant)
	        translate(descendant)
	        if descendant:IsA("TextLabel") or descendant:IsA("TextButton") or descendant:IsA("TextBox") then
	            descendant:GetPropertyChangedSignal("Text"):Connect(function()
	                translate(descendant)
	            end)
	        end
	    end)
	end
	pcall(function() setupSilentTranslator(CoreGui) end)
	pcall(function() setupSilentTranslator(PlayerGui) end)
	PlayerGui.ChildAdded:Connect(function(child)
	    if child:IsA("GuiObject") then
	        setupSilentTranslator(child)
	    end
	end)
